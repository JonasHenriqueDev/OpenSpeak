cmake_minimum_required(VERSION 3.25)
project(OpenSpeakPortable LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT EXISTS "${PAYLOAD_ZIP}")
    message(FATAL_ERROR "PAYLOAD_ZIP must point to the packaged portable ZIP")
endif()
get_filename_component(APP_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../client/assets/openspeak.ico" ABSOLUTE)
file(TO_CMAKE_PATH "${PAYLOAD_ZIP}" PAYLOAD_ZIP)
configure_file(payload.rc.in payload.rc @ONLY)
add_executable(OpenSpeakPortable WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
target_compile_definitions(OpenSpeakPortable PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(OpenSpeakPortable PRIVATE /utf-8)
target_link_libraries(OpenSpeakPortable PRIVATE ole32)
